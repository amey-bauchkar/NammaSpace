#   tools/uv/uv.exe venv --python 3.10 tools/splat-env
#   tools/uv/uv.exe pip install --python tools/splat-env/Scripts/python.exe torch==2.4.1 --index-url https://download.pytorch.org/whl/cu124
#   tools/uv/uv.exe pip install --python tools/splat-env/Scripts/python.exe -r requirements-splat.txt --find-links https://docs.gsplat.studio/whl/gsplat/
#
